"""Saves every telemetry line from the board to a CSV file (for the report plots).

Each row = one $TEL message (50 per second) plus what the Pi was doing at that
moment: the target it was sending, and whether it was sending at all.
Scripts can add their own columns (the camera adds cam_y, found, ...).
"""

import csv
import os
import time

import protocol

LOG_DIR = os.path.join(os.path.dirname(os.path.abspath(__file__)), "logs")


class DataLog:
    def __init__(self, link, name, extra=None):
        """link  = BoardLink to record from
        name  = start of the file name, e.g. "step"
        extra = optional function returning a dict of extra columns"""
        os.makedirs(LOG_DIR, exist_ok=True)
        stamp = time.strftime("%Y%m%d_%H%M%S")
        self.path = os.path.join(LOG_DIR, "%s_%s.csv" % (name, stamp))
        self.link = link
        self.extra = extra
        self.t0 = time.time()
        self.rows = 0
        self._file = open(self.path, "w", newline="")
        self._writer = None
        link.listeners.append(self._on_tel)

    def _on_tel(self, tel):
        row = {"t": round(tel["pi_time"] - self.t0, 3)}
        row.update({k: tel[k] for k in protocol.TEL_FIELDS})
        row["pi_target"] = self.link.target
        row["pi_sending"] = int(self.link.sending)
        if self.extra:
            row.update(self.extra())
        if self._writer is None:        # header from the first row
            self._writer = csv.DictWriter(self._file, fieldnames=list(row))
            self._writer.writeheader()
        self._writer.writerow(row)
        self.rows += 1

    def close(self):
        if self._on_tel in self.link.listeners:
            self.link.listeners.remove(self._on_tel)
        self._file.close()
        return self.path

"""Automatic test for the report: the Pi changes the target in steps while the
board keeps the rig balanced, then stops sending to show the link-loss safety.
Everything is saved to logs/step_<date>_<time>.csv.

Run:  python3 step_test.py      (press K1 first so the motors are armed)
Ctrl+C stops early (the file is still saved, and the board returns to level).
"""

import time

from datalog import DataLog
from link import BoardLink

# (what to do, target in degrees, how long in seconds)
PLAN = [
    ("hold", 0.0, 4),
    ("hold", 10.0, 6),
    ("hold", 0.0, 6),
    ("hold", -10.0, 6),
    ("hold", 0.0, 5),
    ("hold", 10.0, 5),
    ("stop", 10.0, 4),      # Pi goes silent -> board must return to 0 by itself
    ("hold", 0.0, 4),       # sending again
]


def wait_for_board(link):
    print("Waiting for telemetry from the board...")
    while link.latest is None:
        time.sleep(0.1)
    if not link.latest["armed"]:
        print("Motors are NOT armed. Press K1, wait for the countdown beeps...")
        while not link.latest["armed"]:
            time.sleep(0.1)
    print("Armed. Starting in 3 s - keep the power switch within reach.")
    time.sleep(3)


def main():
    link = BoardLink()
    link.set_target(0.0)
    try:
        wait_for_board(link)
        log = DataLog(link, "step")
        total = sum(p[2] for p in PLAN)
        print("Test takes %d s. Saving to %s" % (total, log.path))
        for action, target, seconds in PLAN:
            link.set_target(target)
            link.sending = (action != "stop")
            if action == "stop":
                print(">> Pi STOPS sending for %d s (rig should go back to level)" % seconds)
            else:
                print(">> target %+.0f deg for %d s" % (target, seconds))
            end = time.time() + seconds
            while time.time() < end:
                tel = link.latest
                print("   pitch %6.1f  board target %5.1f  link %s"
                      % (tel["pitch"], tel["target"], "LOST" if tel["link_lost"] else "ok"),
                      end="\r")
                time.sleep(0.2)
            print()
        link.set_target(0.0)
        link.sending = True
        time.sleep(1)
        print("Done: %d rows saved to %s" % (log.rows, log.close()))
        print("Make the plot:  python3 plot_log.py %s" % log.path)
    except KeyboardInterrupt:
        print("\nStopped early.")
        try:
            print("Saved %d rows to %s" % (log.rows, log.close()))
        except NameError:
            pass
    link.close()


if __name__ == "__main__":
    main()

"""Makes the report plot from a log file.

Run:  python3 plot_log.py logs/step_20261113_160000.csv
      python3 plot_log.py            (no name = newest file in logs/)
Saves a .png next to the .csv and opens it on screen.
"""

import csv
import glob
import os
import sys

try:
    import matplotlib.pyplot as plt
except ImportError:
    sys.exit("matplotlib missing. Install it with:  sudo apt install -y python3-matplotlib")

from datalog import LOG_DIR


def load(path):
    with open(path) as f:
        rows = list(csv.DictReader(f))
    if not rows:
        sys.exit("The file is empty: " + path)
    return {k: [float(r[k]) for r in rows] for k in rows[0]}


def shade(ax, t, flags, color, label):
    """Shades the time ranges where flags == 1."""
    start = None
    for i, f in enumerate(flags + [0]):
        if f and start is None:
            start = t[i]
        elif not f and start is not None:
            ax.axvspan(start, t[i - 1], color=color, alpha=0.15, label=label)
            label = None
            start = None


def main():
    if len(sys.argv) > 1:
        path = sys.argv[1]
    else:
        files = sorted(glob.glob(os.path.join(LOG_DIR, "*.csv")), key=os.path.getmtime)
        if not files:
            sys.exit("No log files in " + LOG_DIR)
        path = files[-1]
    d = load(path)
    t = d["t"]
    camera = "cam_y" in d
    n = 3 if camera else 2
    fig, axes = plt.subplots(n, 1, sharex=True, figsize=(10, 3 * n))

    ax = axes[0]
    sent = [v if s else float("nan") for v, s in zip(d["pi_target"], d["pi_sending"])]
    ax.plot(t, sent, "--", color="grey", label="Pi sends")
    ax.plot(t, d["target"], color="black", label="board target")
    ax.plot(t, d["pitch"], label="pitch (IMU)")
    ax.plot(t, d["enc_pitch"], alpha=0.7, label="pitch (encoder)")
    shade(ax, t, [int(v) for v in d["link_lost"]], "red", "link lost")
    ax.set_ylabel("angle (deg)")
    ax.set_title("Raspberry Pi target vs rig pitch")
    ax.legend(loc="upper right", fontsize=8)
    ax.grid(True)

    ax = axes[1]
    ax.plot(t, d["left_us"], label="left motors")
    ax.plot(t, d["right_us"], label="right motors")
    ax.set_ylabel("ESC pulse (us)")
    ax.legend(loc="upper right", fontsize=8)
    ax.grid(True)

    if camera:
        ax = axes[2]
        ax.plot(t, d["cam_y"], label="object height in image (-1 bottom, +1 top)")
        shade(ax, t, [1 - int(v) for v in d["found"]], "orange", "object not seen")
        ax.set_ylabel("camera")
        ax.legend(loc="upper right", fontsize=8)
        ax.grid(True)

    axes[-1].set_xlabel("time (s)")
    fig.tight_layout()
    png = os.path.splitext(path)[0] + ".png"
    fig.savefig(png, dpi=150)
    print("Saved", png)
    plt.show()


if __name__ == "__main__":
    main()

cat > camera_track.py <<'PYEOF'
"""Milestone 3 camera part: the Pi finds a coloured object and turns its
height in the image into the rig's target angle. The board still does all the
balancing; the Pi only sends the target ($SET), like console.py.

Run:  python3 camera_track.py               (object higher in image = rig tilts up)
      python3 camera_track.py --follow      (camera ON the rig: rig tilts to keep
                                             the object in the middle of the image)
      python3 camera_track.py --no-board    (camera only, to tune the colour)

Keys (click the camera window first):
  s = start / pause following (starts PAUSED: target stays 0)
  f = flip direction (if the rig tilts the wrong way)
  q = quit
The sliders in the "colour" window pick the colour (HSV range).
Every run is saved to logs/camera_<date>_<time>.csv.
"""

import argparse
import time

import cv2
import numpy as np

WIDTH, HEIGHT = 640, 480
MAX_TARGET = 15.0       # deg, the Pi never asks for more (board limit is 20)
DEADBAND = 0.10         # ignore small movements near the middle of the image
SMOOTH = 0.15           # 0..1, smaller = smoother but slower target changes
FOLLOW_SPEED = 12.0     # deg/s at full error, for --follow
LOST_TIME = 0.5         # s without the object -> target goes back to 0
MIN_AREA = 400          # pixels, smaller blobs are ignored as noise
ROTATE_180 = False      # set True if the picture is upside down

# starting colour range (HSV). Tune with the sliders, then copy the numbers here.
# OpenCV hue is 0-179: red ~0-10/170-179, orange ~10-25, yellow ~25-35,
# green ~40-80, blue ~100-130.
HSV_LOW = (100, 120, 70)
HSV_HIGH = (130, 255, 255)


class Camera:
    """Picamera2 on the Pi; falls back to a USB webcam (cv2) elsewhere."""

    def __init__(self):
        try:
            from picamera2 import Picamera2
            self.cam = Picamera2()
            # "RGB888" gives the bytes in BGR order, which is what OpenCV wants
            cfg = self.cam.create_preview_configuration(
                main={"size": (WIDTH, HEIGHT), "format": "RGB888"})
            self.cam.configure(cfg)
            self.cam.start()
            self.pi = True
        except ImportError:
            self.cam = cv2.VideoCapture(0)
            self.pi = False

    def read(self):
        if self.pi:
            frame = self.cam.capture_array()
        else:
            ok, frame = self.cam.read()
            if not ok:
                return None
            frame = cv2.resize(frame, (WIDTH, HEIGHT))
        if ROTATE_180:
            frame = cv2.rotate(frame, cv2.ROTATE_180)
        return frame

    def close(self):
        if self.pi:
            self.cam.stop()
        else:
            self.cam.release()


def make_sliders():
    cv2.namedWindow("colour")
    names = ["H low", "S low", "V low", "H high", "S high", "V high"]
    start = list(HSV_LOW) + list(HSV_HIGH)
    for name, value in zip(names, start):
        cv2.createTrackbar(name, "colour", value, 179 if name[0] == "H" else 255,
                           lambda v: None)


def read_sliders():
    g = lambda n: cv2.getTrackbarPos(n, "colour")
    return (np.array([g("H low"), g("S low"), g("V low")]),
            np.array([g("H high"), g("S high"), g("V high")]))


def find_object(frame, low, high):
    """Returns (mask, (x, y, radius) of the biggest blob or None)."""
    hsv = cv2.cvtColor(cv2.GaussianBlur(frame, (7, 7), 0), cv2.COLOR_BGR2HSV)
    if low[0] <= high[0]:
        mask = cv2.inRange(hsv, low, high)
    else:   # hue range that wraps around (red): e.g. low 170, high 10
        mask = cv2.inRange(hsv, low, np.array([179, high[1], high[2]])) | \
               cv2.inRange(hsv, np.array([0, low[1], low[2]]), high)
    mask = cv2.dilate(cv2.erode(mask, None, iterations=2), None, iterations=2)
    contours, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    if not contours:
        return mask, None
    biggest = max(contours, key=cv2.contourArea)
    if cv2.contourArea(biggest) < MIN_AREA:
        return mask, None
    (x, y), r = cv2.minEnclosingCircle(biggest)
    return mask, (int(x), int(y), int(r))


class Tracker:
    """Turns the object position into a target angle."""

    def __init__(self, follow):
        self.follow = follow
        self.active = False         # 's' key
        self.sign = 1.0             # 'f' key
        self.target = 0.0
        self.cam_y = 0.0            # -1 bottom .. +1 top of the image
        self.found = False
        self.last_seen = 0.0
        self.last_time = time.time()

    def update(self, obj):
        now = time.time()
        dt = now - self.last_time
        self.last_time = now
        self.found = obj is not None
        if self.found:
            self.last_seen = now
            self.cam_y = (HEIGHT / 2 - obj[1]) / (HEIGHT / 2)

        lost = now - self.last_seen > LOST_TIME
        if not self.active or lost:
            wanted = 0.0                    # safe: back to level
            if self.follow:
                self.target += (0.0 - self.target) * SMOOTH
                return self.target
        else:
            err = self.cam_y if abs(self.cam_y) > DEADBAND else 0.0
            if self.follow:
                # keep moving the rig until the object is in the middle
                self.target += self.sign * FOLLOW_SPEED * err * dt
                self.target = max(-MAX_TARGET, min(MAX_TARGET, self.target))
                return self.target
            wanted = self.sign * err * MAX_TARGET
        self.target += (wanted - self.target) * SMOOTH
        return self.target


def draw(frame, mask, obj, tr, tel, fps):
    h, w = frame.shape[:2]
    cv2.line(frame, (0, h // 2), (w, h // 2), (255, 255, 255), 1)
    band = int(DEADBAND * h / 2)
    cv2.line(frame, (0, h // 2 - band), (w, h // 2 - band), (128, 128, 128), 1)
    cv2.line(frame, (0, h // 2 + band), (w, h // 2 + band), (128, 128, 128), 1)
    if obj:
        x, y, r = obj
        cv2.circle(frame, (x, y), r, (0, 255, 0), 2)
        cv2.circle(frame, (x, y), 4, (0, 0, 255), -1)
        cv2.line(frame, (x, h // 2), (x, y), (0, 255, 255), 2)
    # small black-and-white mask in the top-right corner
    small = cv2.resize(cv2.cvtColor(mask, cv2.COLOR_GRAY2BGR), (w // 4, h // 4))
    frame[0:h // 4, w - w // 4:w] = small

    state = "FOLLOWING" if tr.active else "PAUSED (press s)"
    lines = ["%s   mode: %s   dir %+d" % (state, "follow" if tr.follow else "height", tr.sign),
             "object: %s   y %+.2f" % ("found" if tr.found else "NOT SEEN", tr.cam_y),
             "Pi target: %+.1f deg" % tr.target]
    if tel:
        lines.append("board: target %+.1f  pitch %+.1f  armed %d  link %s"
                     % (tel["target"], tel["pitch"], tel["armed"],
                        "LOST" if tel["link_lost"] else "ok"))
    else:
        lines.append("board: not connected")
    lines.append("%.0f fps" % fps)
    panel = frame[0:22 * len(lines) + 8, 0:w - w // 4]
    panel[:] = (panel * 0.4).astype(np.uint8)       # darken behind the text
    for i, text in enumerate(lines):
        cv2.putText(frame, text, (8, 22 + 22 * i), cv2.FONT_HERSHEY_SIMPLEX, 0.55,
                    (0, 255, 0) if tr.active else (0, 200, 255), 1, cv2.LINE_AA)


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--follow", action="store_true", help="camera is on the rig")
    ap.add_argument("--no-board", action="store_true", help="camera only")
    args = ap.parse_args()

    link = log = None
    tr = Tracker(args.follow)
    if not args.no_board:
        from datalog import DataLog
        from link import BoardLink
        link = BoardLink()
        link.set_target(0.0)
        log = DataLog(link, "camera", extra=lambda: {
            "cam_y": round(tr.cam_y, 3), "found": int(tr.found),
            "following": int(tr.active)})
        print("Board on %s, saving to %s" % (link.port, log.path))

    cam = Camera()
    make_sliders()
    fps, last = 0.0, time.time()
    try:
        while True:
            frame = cam.read()
            if frame is None:
                break
            low, high = read_sliders()
            mask, obj = find_object(frame, low, high)
            target = tr.update(obj)
            if link:
                link.set_target(round(target, 1))
            now = time.time()
            fps = 0.9 * fps + 0.1 / max(now - last, 1e-3)
            last = now
            draw(frame, mask, obj, tr, link.latest if link else None, fps)
            cv2.imshow("camera", frame)
            key = cv2.waitKey(1) & 0xFF
            if key == ord("q"):
                break
            elif key == ord("s"):
                tr.active = not tr.active
                print(">> following" if tr.active else ">> paused (target back to 0)")
            elif key == ord("f"):
                tr.sign = -tr.sign
                print(">> direction flipped")
    except KeyboardInterrupt:
        pass
    low, high = read_sliders()
    print("Colour used: HSV_LOW = (%d, %d, %d), HSV_HIGH = (%d, %d, %d)"
          % (tuple(low) + tuple(high)))
    cam.close()
    cv2.destroyAllWindows()
    if link:
        link.set_target(0.0)
        time.sleep(0.3)
        print("Saved %d rows to %s" % (log.rows, log.close()))
        link.close()


if __name__ == "__main__":
    main()
